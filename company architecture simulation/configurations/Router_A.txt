enable
configure terminal
hostname Router_A

! --- Inter-VLAN Routing (Router-on-a-Stick) ---
interface gig0/0
 no shutdown
interface gig0/0.10
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
 ip nat inside
interface gig0/0.20
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
 ip nat inside
interface gig0/0.30
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
 ip nat inside
interface gig0/0.40
 encapsulation dot1Q 40
 ip address 192.168.40.1 255.255.255.0
 ip nat inside
interface gig0/0.50
 encapsulation dot1Q 50
 ip address 192.168.50.1 255.255.255.0
 ip nat inside

! --- WAN Interfaces ---
interface gig0/1
 ip address 10.0.0.1 255.255.255.252
 ip nat inside
 no shutdown
interface gig0/2
 ip address 200.1.1.2 255.255.255.0
 ip nat outside
 no shutdown

! --- OSPF Routing ---
router ospf 1
 router-id 1.1.1.1
 network 192.168.10.0 0.0.0.255 area 0
 network 192.168.20.0 0.0.0.255 area 0
 network 192.168.30.0 0.0.0.255 area 0
 network 192.168.40.0 0.0.0.255 area 0
 network 192.168.50.0 0.0.0.255 area 0
 network 10.0.0.0 0.0.0.3 area 0
 default-information originate

! --- Default Route to ISP ---
ip route 0.0.0.0 0.0.0.0 200.1.1.1

! --- NAT/PAT ---
access-list 1 permit 192.168.0.0 0.0.255.255
ip nat inside source list 1 interface gig0/2 overload

! --- ACL 100: Guest Isolation ---
access-list 100 deny ip 192.168.40.0 0.0.0.255 192.168.0.0 0.0.255.255
access-list 100 permit udp 192.168.40.0 0.0.0.255 host 192.168.10.10 eq 53
access-list 100 permit tcp 192.168.40.0 0.0.0.255 any eq 80
access-list 100 permit tcp 192.168.40.0 0.0.0.255 any eq 443
access-list 100 permit ip any any
interface gig0/0.40
 ip access-group 100 in

! --- ACL 110: Finance Protection ---
access-list 110 permit ip 192.168.10.0 0.0.0.255 192.168.20.0 0.0.0.255
access-list 110 permit ip 192.168.30.0 0.0.0.255 192.168.20.0 0.0.0.255
access-list 110 deny ip any 192.168.20.0 0.0.0.255 log
access-list 110 permit ip any any
interface gig0/0.20
 ip access-group 110 in

! --- SSH Configuration (ACL 120) ---
ip domain-name techglobal.lan
crypto key generate rsa

username admin privilege 15 secret admin123

access-list 120 deny tcp any any eq 22
line vty 0 4
 access-class 120 in
 login local
 transport input ssh
end
